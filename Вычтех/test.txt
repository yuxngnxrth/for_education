// ==============================================
// Программа для вычисления out = (k*a + b) + c
// где: k = sw_i[15:12], a = sw_i[11:8]
//       b = sw_i[7:4], c = sw_i[3:0]
// ==============================================

// Инициализация регистров
// J B WS  const23 (23 бита)         WA
  0 0 00 00000000000000000000000 00010 // reg2 = 0 (результат)
  0 0 00 00000000000000000001111 00011 // reg3 = 15 (маска 0xF)
  0 0 00 00000000000000000000100 00100 // reg4 = 4 (для сдвигов)
  0 0 00 00000000000000000000001 00101 // reg5 = 1 (инкремент/декремент)

// Загрузка sw_i в регистр 1
// J B WS ALUop  RA1   RA2   const     WA
  0 0 10 00000 00000 00000 00000000 00001 // reg1 = sw_i

// ==============================================
// Извлечение k (биты [15:12])
// ==============================================
// Сдвиг reg1 на 12 бит вправо (3 раза по 4)
  0 0 01 00101 00001 00100 00000000 00110 // reg6 = reg1 >> 4
  0 0 01 00101 00110 00100 00000000 00110 // reg6 = reg6 >> 4
  0 0 01 00101 00110 00100 00000000 00110 // reg6 = reg6 >> 4

// Применение маски 0xF
  0 0 01 00111 00110 00011 00000000 00110 // reg6 = reg6 AND reg3

// Сохранение k в reg7
  0 0 01 00000 00110 00000 00000000 00111 // reg7 = reg6

// ==============================================
// Извлечение a (биты [11:8])
// ==============================================
// Сдвиг reg1 на 8 бит вправо (2 раза по 4)
  0 0 01 00101 00001 00100 00000000 00110 // reg6 = reg1 >> 4
  0 0 01 00101 00110 00100 00000000 00110 // reg6 = reg6 >> 4

// Применение маски 0xF
  0 0 01 00111 00110 00011 00000000 00110 // reg6 = reg6 AND reg3

// Сохранение a в reg8 (будет счетчиком в цикле)
  0 0 01 00000 00110 00000 00000000 01000 // reg8 = reg6

// ==============================================
// Цикл умножения k * a (складываем k a раз)
// ==============================================
// Метка: start_loop
  0 1 00 11000 01000 00000 00000100 00000 // if reg8 == 0, skip 4 инструкции

// Тело цикла:
  0 0 01 00000 00010 00111 00000000 00010 // reg2 = reg2 + reg7 (добавляем k)
  0 0 01 01000 01000 00101 00000000 01000 // reg8 = reg8 - 1 (уменьшаем счетчик)

  1 0 00 00000 00000 00000 11111101 00000 // jump -3 (назад к start_loop)

// ==============================================
// Извлечение b (биты [7:4])
// ==============================================
// Сдвиг reg1 на 4 бита вправо
  0 0 01 00101 00001 00100 00000000 00110 // reg6 = reg1 >> 4

// Применение маски 0xF
  0 0 01 00111 00110 00011 00000000 00110 // reg6 = reg6 AND reg3

// Добавление b к результату
  0 0 01 00000 00010 00110 00000000 00010 // reg2 = reg2 + reg6

// ==============================================
// Извлечение c (биты [3:0])
// ==============================================
// Применение маски 0xF к reg1 (без сдвига)
  0 0 01 00111 00001 00011 00000000 00110 // reg6 = reg1 AND reg3

// ==============================================
// Цикл умножения k * a + b (складываем (k * a + b) c раз)
// ==============================================
// Метка: start_loop
  0 1 00 11000 00011 00000 00000100 00000 // if reg6 == 0, skip 4 инструкции

// Тело цикла:
  0 0 01 00000 00010 00010 00000000 00010 // reg2 = reg2 + reg2 (добавляем (k * a + b) )
  0 0 01 01000 00011 00101 00000000 00011 // reg6 = reg6 - 1 (уменьшаем счетчик)

  1 0 00 00000 00000 00000 11111101 00000 // jump -3 (назад к start_loop)

// ==============================================
// Бесконечный цикл вывода результата
// ==============================================
  1 0 00 00000 00010 00000 00000000 00000 // Бесконечный цикл, вывод reg2
